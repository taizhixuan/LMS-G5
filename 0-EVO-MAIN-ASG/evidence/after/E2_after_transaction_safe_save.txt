scenario the proposal Table 1.1 row 3 cites. The E2 fix is
preventative; the negative reproducer here does not invalidate the

Code-level proof — the actual change in librarian/borrow_save.php
======================================================================
Pre-E2:
  mysqli_query($con, "INSERT INTO borrow ...");
  $query = mysqli_query($con, "SELECT * FROM borrow ORDER BY borrow_id DESC");
  $row = mysqli_fetch_array($query);
  $borrow_id = $row['borrow_id'];     // race-prone
  foreach ($book_ids as $bid) {
      mysqli_query($con, "INSERT INTO borrowdetails ...");
  }

Post-E2:
  mysqli_begin_transaction($con);
  try {
      $stmt = mysqli_prepare($con, "INSERT INTO borrow ...");
      mysqli_stmt_bind_param($stmt, 'is', $member_id, $due_date_sql);
      if (!mysqli_stmt_execute($stmt)) throw new Exception('...');
      $borrow_id = mysqli_insert_id($con);   // per-connection, race-free
      mysqli_stmt_close($stmt);

      $stmt = mysqli_prepare($con, "INSERT INTO borrowdetails ...");
      foreach ($book_ids as $bid) {
          mysqli_stmt_bind_param($stmt, 'ii', $bid, $borrow_id);
          if (!mysqli_stmt_execute($stmt)) throw new Exception('...');
      }
      mysqli_stmt_close($stmt);
      mysqli_commit($con);
  } catch (Exception $e) {
      mysqli_rollback($con);
      fail('Could not save borrow transaction: ' . $e->getMessage());
  }
