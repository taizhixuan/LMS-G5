Proves: borrow and borrowdetails tables ship as MyISAM (no transaction support,
no foreign keys), and store dates as varchar(100). The eb_lms.sql seed data
already mixes two date formats:
  - date_borrow uses '2014-03-20 23:50:27' (ISO-ish)
  - due_date    uses '21/03/2014'          (DD/MM/YYYY)
  - date_return uses '2014-03-21 00:30:51' for returned rows, '' for pending
The E3 migration must use STR_TO_DATE with both patterns or rows will null out.

======================================================================
borrow table
======================================================================
CREATE TABLE `borrow` (
  `borrow_id`   int(11)      NOT NULL AUTO_INCREMENT,
  `member_id`   bigint(50)   NOT NULL,
  `date_borrow` varchar(100) NOT NULL,        -- should be DATETIME
  `due_date`    varchar(100) DEFAULT NULL,    -- should be DATE/DATETIME
  PRIMARY KEY (`borrow_id`),
  KEY `borrowerid` (`member_id`),
  KEY `borrowid`   (`borrow_id`)
) ENGINE=MyISAM AUTO_INCREMENT=485 DEFAULT CHARSET=utf8 COLLATE=utf8_general_ci
                                            -- should be InnoDB; missing FK to member

======================================================================
borrowdetails table
======================================================================
CREATE TABLE `borrowdetails` (
  `borrow_details_id` int(11)      NOT NULL AUTO_INCREMENT,
  `book_id`           int(11)      NOT NULL,
  `borrow_id`         int(11)      NOT NULL,
  `borrow_status`     varchar(50)  NOT NULL,
  `date_return`       varchar(100) NOT NULL,  -- should be DATETIME NULL
  PRIMARY KEY (`borrow_details_id`)
) ENGINE=MyISAM AUTO_INCREMENT=165 DEFAULT CHARSET=latin1 COLLATE=latin1_swedish_ci
                                            -- should be InnoDB; missing FKs to book + borrow

book table (already InnoDB — does NOT need engine change in E3)
======================================================================
CREATE TABLE `book` (
  `book_id`        int(11)      NOT NULL AUTO_INCREMENT,
  `book_title`     varchar(100) NOT NULL,
  `category_id`    int(50)      NOT NULL,
  `author`         varchar(50)  NOT NULL,
  `book_copies`    int(11)      NOT NULL,
  `book_pub`       varchar(100) NOT NULL,
  `publisher_name` varchar(100) NOT NULL,
  `isbn`           varchar(50)  NOT NULL,
  `copyright_year` int(11)      NOT NULL,
  `date_receive`   varchar(20)  NOT NULL,
  `date_added`     datetime     NOT NULL,
  `status`         varchar(30)  NOT NULL,
  PRIMARY KEY (`book_id`)
) ENGINE=InnoDB AUTO_INCREMENT=33 DEFAULT CHARSET=latin1 COLLATE=latin1_swedish_ci
